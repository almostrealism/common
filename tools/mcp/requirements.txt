# Consolidated requirements for all AR MCP servers and their tests
# Install with: pip install -r tools/mcp/requirements.txt
#
# This is the file to install. Each server's own requirements.txt is a subset
# of it, kept for building that server alone (its Dockerfile or start.sh); a package a
# server needs belongs here as well as there.
#
# Covers: ar-manager, ar-memory, ar-test-runner, ar-build-validator, ar-jmx,
#         ar-profile-analyzer, ar-commit, the shared library in common/, and
#         the Python tests of all of them.
#
# Python 3.10 through 3.14. CI runs 3.10. Python 3.15 cannot install this
# file yet: fastembed needs onnxruntime, which publishes no 3.15 wheels, and
# pip reports that as "ResolutionImpossible" on fastembed rather than naming
# the interpreter. Use a 3.14 or older interpreter for the virtualenv.

# Core MCP framework (all servers)
# Held below 2.0.0: the 2.x SDK removes mcp.server.fastmcp, which every
# decorator-based server here imports. Raising this bound requires migrating
# the servers to the 2.x API first.
mcp>=1.0.0,<2

# Embeddings and vector search (ar-memory)
fastembed>=0.2.0
faiss-cpu>=1.7.0
numpy

# HTTP server (ar-memory HTTP API, ar-manager)
starlette>=0.27.0
uvicorn>=0.23.0

# Imported directly by tests: httpx by the ar-manager transport tests, pytest
# by the ar-jmx and ar-test-runner tests, pyyaml by the workstream-migration
# and CI-tool tests under tools/ that run with this environment.
httpx
pytest
pyyaml
